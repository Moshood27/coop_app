92:     <script>
93:         if (!Alpine.data('biometricStationHandler')) {
94:             Alpine.data('biometricStationHandler', () => ({
95:                 processing: false,
96:
97:                 async handleRegistration(options) {
98:                     this.processing = true;
99:                     try {
100:                         const publicKey = this.parseOptions(options);
101:                         const credential = await navigator.credentials.create({ publicKey });
102:                         const credentialJSON = this.publicKeyCredentialToJSON(credential);
103:
104:                         $wire.completeEnrollment(credentialJSON);
105:                     } catch (e) {
106:                         console.error('Registration failed', e);
107:                         this.processing = false;
108:                     }
109:                 },
110:
111:                 async handleVerification(options) {
112:                     this.processing = true;
113:                     try {
114:                         const publicKey = this.parseOptions(options);
115:                         const assertion = await navigator.credentials.get({ publicKey });
116:                         const assertionJSON = this.publicKeyCredentialToJSON(assertion);
117:
118:                         $wire.completeVerification(assertionJSON);
119:                     } catch (e) {
120:                         console.error('Verification failed', e);
121:                         this.processing = false;
122:                     }
123:                 },
124:
125:                 bufferToBase64url(buffer) {
126:                     const byteView = new Uint8Array(buffer);
127:                     let str = '';
128:                     for (const charCode of byteView) {
129:                         str += String.fromCharCode(charCode);
130:                     }
131:                     return btoa(str)
132:                         .replace(/\+/g, '-')
133:                         .replace(/\//g, '_')
134:                         .replace(/=/g, '');
135:                 },
136:
137:                 base64urlToBuffer(base64url) {
138:                     const base64 = base64url.replace(/-/g, '+').replace(/_/g, '/');
139:                     const padLen = (4 - (base64.length % 4)) % 4;
140:                     const str = atob(base64 + '='.repeat(padLen));
141:                     const buffer = new ArrayBuffer(str.length);
142:                     const byteView = new Uint8Array(buffer);
143:                     for (let i = 0; i < str.length; i++) {
144:                         byteView[i] = str.charCodeAt(i);
145:                     }
146:                     return buffer;
147:                 },
148:
149:                 publicKeyCredentialToJSON(pubKeyCred) {
150:                     if (pubKeyCred instanceof Array) {
151:                         return pubKeyCred.map(p => this.publicKeyCredentialToJSON(p));
152:                     }
153:
154:                     if (pubKeyCred instanceof ArrayBuffer) {
155:                         return this.bufferToBase64url(pubKeyCred);
156:                     }
157:
158:                     if (pubKeyCred instanceof Object) {
159:                         const obj = {};
160:                         for (const key in pubKeyCred) {
161:                             if (typeof pubKeyCred[key] === 'function') continue;
162:
163:                             if (pubKeyCred[key] instanceof ArrayBuffer) {
164:                                 obj[key] = this.bufferToBase64url(pubKeyCred[key]);
165:                             } else if (typeof pubKeyCred[key] === 'object' && pubKeyCred[key] !== null) {
166:                                 obj[key] = this.publicKeyCredentialToJSON(pubKeyCred[key]);
167:                             } else {
168:                                 obj[key] = pubKeyCred[key];
169:                             }
170:                         }
171:
172:                         if (pubKeyCred.getClientExtensionResults) {
173:                             obj.clientExtensionResults = pubKeyCred.getClientExtensionResults();
174:                         }
175:
176:                         return obj;
177:                     }
178:
179:                     return pubKeyCred;
180:                 },
181:
182:                 parseOptions(options) {
183:                     const cloned = JSON.parse(JSON.stringify(options));
184:
185:                     if (cloned.challenge) cloned.challenge = this.base64urlToBuffer(cloned.challenge);
186:                     if (cloned.user && cloned.user.id) cloned.user.id = this.base64urlToBuffer(cloned.user.id);
187:
188:                     if (cloned.allowCredentials) {
189:                         cloned.allowCredentials = cloned.allowCredentials.map(cred => ({
190:                             ...cred,
191:                             id: this.base64urlToBuffer(cred.id)
192:                         }));
193:                     }
194:
195:                     if (cloned.excludeCredentials) {
196:                         cloned.excludeCredentials = cloned.excludeCredentials.map(cred => ({
197:                             ...cred,
198:                             id: this.base64urlToBuffer(cred.id)
199:                         }));
200:                     }
201:
202:                     return cloned;
203:                 }
204:             }));
205:         }
